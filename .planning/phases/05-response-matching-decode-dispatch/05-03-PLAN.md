```typescript
// D-06: Duck-type on "kind" — never instanceof. All six DecodeError variants have a string "kind".
// A valid decoded value will never be an object whose "kind" matches a DecodeError kind string.
const DECODE_ERROR_KINDS = new Set<string>([
  "unexpectedBody",
  "emptyBody",
  "invalidJson",
  "schemaMismatch",
  "bodyReadFailed",
  "custom",
]);

function isDecodeError(v: unknown): v is DecodeError {
  return (
    typeof v === "object" &&
    v !== null &&
    "kind" in v &&
    typeof (v as Record<string, unknown>)["kind"] === "string" &&
    DECODE_ERROR_KINDS.has((v as Record<string, unknown>)["kind"] as string)
  );
}
```

```typescript
    const maxBytes = clientSpec.diagnostics?.bodyPreviewBytes ?? 8192;
    const status = response.status;
    const headers = response.headers;

    const match = matchResponse(status, spec.responses, clientSpec.responses);
    if (match === null) {
      // RESP-02, PREV-01: stream not yet consumed — use readBodyPreview (streaming, byte-capped)
      const preview = await readBodyPreview(response, maxBytes);
      return { kind: "unhandledStatus", status, headers, preview };
    }

    // RESP-03: buffer full body for decode + preview derivation
    const bytes = await readBytes(response);
    // readBytes() can return { kind: "bodyReadFailed" } — guard before treating as Uint8Array
    if ("kind" in bytes) {
      const preview = previewFromBytes(new Uint8Array(0), maxBytes);
      return { kind: "decodeError", status, headers, error: bytes, preview };
    }

    const syntheticResponse = new Response(bytes);
    let decoded: unknown;
    try {
      decoded = await match.decode.fn(syntheticResponse);
    } catch (e: unknown) {
      // D-06: decoder threw unexpectedly — wrap to preserve no-throw contract
      const preview = previewFromBytes(bytes, maxBytes);
      return {
        kind: "decodeError",
        status,
        headers,
        error: { kind: "bodyReadFailed", message: String(e) },
        preview,
      };
    }

    if (isDecodeError(decoded)) {
      // RESP-04: decoder returned a DecodeError value — include preview from buffered bytes
      const preview = previewFromBytes(bytes, maxBytes);
      return { kind: "decodeError", status, headers, error: decoded, preview };
    }

    // as unknown as R: the single principled cast in the dispatch path (D-07)
    // Only reachable via TaggedEntry<T> whose T is R's body — double-cast required by
    // no-unsafe-type-assertion rule (no direct as R from unknown).
    return { kind: "response", response: { tag: match.tag, body: decoded } as unknown as R };
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| fetch API → send.ts dispatch | `response.status` and `response.headers` come from the network; `status` is an integer from the fetch API — not user-controlled input |
| Decoder → send.ts | `decode.fn()` may throw unexpectedly; the inner try/catch enforces the no-throw contract |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-03-01 | Denial of Service | `readBytes()` buffers full body | accept | Body size limit is caller's responsibility (fetch itself or a proxy). `readBytes()` accumulates the entire response body — a pathologically large response could consume memory. Out of scope for v1; no user-controlled size limit is specified in docs/SPEC.md. |
| T-05-03-02 | Information Disclosure | `decodeError.preview` exposes partial response body | accept | Preview bytes come from the server response, not user input. The caller explicitly receives this in `SendResult` — it is intentional. `maxBytes` cap (default 8192) limits exposure. |
| T-05-03-03 | Elevation of Privilege | `isDecodeError` DECODE_ERROR_KINDS set membership | mitigate | Guard checks all six known kinds (not just `"kind" in v`) — prevents a decoded object with an arbitrary `kind` field from being misclassified as a `DecodeError`. The six-element set is exhaustive against the `DecodeError` union in `types.ts`. |
| T-05-03-04 | Tampering | `as unknown as R` cast | accept | The cast is only reachable after matching a `TaggedEntry<T>` via `matchResponse` — the `tag` and `body` values originate from the server response decoded by the registered decoder. This is a type-system gap, not a security boundary. |
</threat_model>
