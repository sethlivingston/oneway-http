```typescript
export interface TaggedEntry<T = unknown, Tag extends string = string> {
  readonly tag: Tag;
  readonly phantom?: T;
  readonly decode: DecoderLike;
}

export type ResponseMap = Partial<Record<StatusMatcher, TaggedEntry>>;
```

`"2xx"` is a member of `StatusMatcher`, so `classOf` return value can key into `ResponseMap`.

```typescript
// src/response-matching.ts — response map lookup; pure function, no I/O
// Dependency direction: (consumers) → response-matching.ts → types.ts

import type { ResponseMap, StatusMatcher, TaggedEntry } from "./types.js";

// classOf: maps an HTTP status code to its class matcher key ("1xx"–"5xx").
// Not exported — internal helper only.
// Cast to StatusMatcher required: template literals are typed as string, not StatusMatcher.
function classOf(status: number): StatusMatcher {
  return `${Math.floor(status / 100)}xx` as StatusMatcher;
}

// matchResponse: four-step two-tier precedence lookup (RESP-01).
// Maps are NEVER pre-merged — the two-tier precedence (request over client) is enforced by
// consulting requestMap first (exact then class) before clientMap (exact then class).
// Returns null when no map entry matches the status or its class (RESP-02).
//
// noUncheckedIndexedAccess: all map[key] accesses return TaggedEntry | undefined;
// every result is guarded with !== undefined before returning.
export function matchResponse(
  status: number,
  requestMap: ResponseMap | undefined,
  clientMap: ResponseMap | undefined,
): TaggedEntry | null {
  const classKey = classOf(status);

  if (requestMap !== undefined) {
    const byExact = requestMap[status];
    if (byExact !== undefined) return byExact;
    const byClass = requestMap[classKey];
    if (byClass !== undefined) return byClass;
  }

  if (clientMap !== undefined) {
    const byExact = clientMap[status];
    if (byExact !== undefined) return byExact;
    const byClass = clientMap[classKey];
    if (byClass !== undefined) return byClass;
  }

  return null;
}
```

**Constraints enforced in this implementation:**
- `classOf` is NOT exported — only `matchResponse` is exported (D-04)
- Maps are consulted sequentially (requestMap, then clientMap), never merged — the only way to enforce two-tier precedence (hard constraint, D-04)
- All four `map[key]` accesses are guarded with `!== undefined` before returning (`noUncheckedIndexedAccess` compliance)
- The cast `as StatusMatcher` on the template literal is the single necessary cast — template literals produce `string`, not `StatusMatcher`; this cast is proven-safe (all valid status codes produce a member of the `StatusMatcher` union)
- Imports: only `import type` from `./types.js` — no other imports (pure utility, no I/O, no decode.ts dependency)
- Explicit return type `: TaggedEntry | null` on exported function (TypeScript Narrows rule)
