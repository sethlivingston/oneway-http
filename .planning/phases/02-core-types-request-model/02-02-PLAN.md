export type Method = "GET" | "HEAD" | "POST" | "PUT" | "PATCH" | "DELETE" | "OPTIONS";

export type QueryValue = string | number | boolean;

export type StatusMatcher = number | "1xx" | "2xx" | "3xx" | "4xx" | "5xx";

export type TaggedEntry<T = unknown, Tag extends string = string> = {
  readonly tag: Tag;
  declare readonly _phantom: T;   // type-only; no runtime slot
  readonly _decode: unknown;
};

export type ResponseMap = Partial<Record<StatusMatcher, TaggedEntry>>;

export type InferResponseUnion<M extends ResponseMap> = {
  [K in keyof M]: M[K] extends TaggedEntry<infer T, infer Tag>
    ? { tag: Tag; body: T }
    : never;
}[keyof M];

export type RequestSpec<Responses extends ResponseMap = ResponseMap> = {
  method: Method;
  path?: readonly (string | number)[];
  absoluteUrl?: string | URL;
  query?: Record<string, QueryValue | readonly QueryValue[] | undefined>;
  headers?: Record<string, string | undefined>;
  body?: unknown;
  responses: Responses;
  retry?: RetryPolicy;
  deadlineMs?: number;
};

```typescript
import type {
  QueryValue,
  RequestSpec,
  ResponseMap,
  InferResponseUnion,
} from "./types.js";

type PathOrUrl =
  | { readonly path: readonly (string | number)[]; readonly absoluteUrl?: undefined }
  | { readonly absoluteUrl: string | URL; readonly path?: undefined };

type RequestInput<Responses extends ResponseMap = ResponseMap> = Omit<
  RequestSpec<Responses>,
  "path" | "absoluteUrl"
> &
  PathOrUrl;

export function buildPath(segments: readonly (string | number)[]): string {
  return segments.map((s) => encodeURIComponent(String(s))).join("/");
}

export function buildQuery(
  query: Record<string, QueryValue | readonly QueryValue[] | undefined>,
): URLSearchParams {
  const params = new URLSearchParams();
  for (const [key, value] of Object.entries(query)) {
    if (value === undefined) continue;
    if (Array.isArray(value)) {
      for (const v of value) params.append(key, String(v));
    } else {
      params.append(key, String(value));
    }
  }
  return params;
}

export class Request<R> {
  #consumed = false;
  readonly #spec: RequestSpec;

  private constructor(spec: RequestSpec) {
    this.#spec = spec;
  }

  static create<M extends ResponseMap>(
    input: RequestInput<M>,
  ): Request<InferResponseUnion<M>> {
    return new Request(input as RequestSpec);
  }

  consume(): RequestSpec {
    if (this.#consumed) {
      throw new TypeError(
        "Request has already been consumed and cannot be sent again",
      );
    }
    this.#consumed = true;
    return this.#spec;
  }
}
```

Key implementation notes:
- **D-08**: `#consumed = false` is a true JS private class field — NOT Symbol, NOT WeakMap.
- **D-07**: The exact TypeError message is `"Request has already been consumed and cannot be sent again"` — do not paraphrase.
- **PathOrUrl union type**: Enforces that callers must provide either `path` OR `absoluteUrl` but not both (compile-time mutual exclusion). The `RequestSpec` stored in `#spec` accepts either via the `as RequestSpec` cast.
- **`buildPath`**: Uses `encodeURIComponent(String(s))` per segment — numbers are stringified before encoding. Joined with `/`. Empty array → empty string (correct behavior for `"".join(/)` behavior).
- **`buildQuery`**: Uses `URLSearchParams` (not manual string concat). `Array.isArray(value)` check handles `readonly QueryValue[]` arrays by iterating and appending each element. Numbers and booleans are stringified via `String(v)`.
- **Import style**: All imports use `import type` (verbatimModuleSyntax) and `.js` extension (ESM). No `import type` is needed for `URLSearchParams` — it's a global.
- **Named exports only**: `export class Request`, `export function buildPath`, `export function buildQuery`. No default export.
- `buildPath` and `buildQuery` are exported from `request.ts` (not re-exported from package entrypoints) so they're accessible to `tests/unit/request.test.ts` and will be used by `src/send.ts` in Phase 3.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Caller input → Request.create() | Caller provides method, path segments, query values, headers — all consumed as-is into RequestSpec |
| path segment encoding → URL | encodeURIComponent() processes caller-controlled path segments before URL construction |
| query value encoding → URLSearchParams | URLSearchParams encodes caller-controlled query values |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-02-01 | Tampering | Path segment injection via unencoded separator characters | mitigate | `encodeURIComponent()` per segment encodes `/`, `?`, `#`, `&` and all RFC 3986 reserved chars. Each segment is encoded independently before joining — a segment containing `/` becomes `%2F`, not a path separator. This is verified by the REQ-02 test `buildPath(["a/b"]) === "a%2Fb"`. |
| T-02-02-02 | Tampering | Query value injection via unencoded special characters | mitigate | `URLSearchParams.append(key, String(value))` encodes all special characters per the `application/x-www-form-urlencoded` spec. No manual string concatenation. |
| T-02-02-03 | Tampering | Affine bypass via private field access | accept | `#consumed` is a true JS private class field (not Symbol, not WeakMap). Private class fields are inaccessible outside the class body at the language level. A caller who reaches into private state through reflection/proxies is violating the API contract — this is a developer error, not a security concern for a library. |
| T-02-02-04 | Information Disclosure | RequestSpec stored in #spec after consume() | accept | After `consume()` returns the RequestSpec, the `#spec` reference is still held internally. This is not a concern — the caller receives the spec and owns it. The Request object itself becomes a spent token with no further capability. |
</threat_model>
