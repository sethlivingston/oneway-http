```typescript
// src/types.ts — shared type definitions; zero imports, zero logic

export type Method =
  | "GET"
  | "HEAD"
  | "POST"
  | "PUT"
  | "PATCH"
  | "DELETE"
  | "OPTIONS";

export type QueryValue = string | number | boolean;

export type StatusMatcher = number | "1xx" | "2xx" | "3xx" | "4xx" | "5xx";

export type Schema<T> = {
  safeParse(
    value: unknown,
  ): { success: true; data: T } | { success: false; error: unknown };
};

export type DecodeIssue = {
  path: Array<string | number>;
  message: string;
  code?: string;
};

export type DecodeError =
  | { kind: "unexpectedBody" }
  | { kind: "emptyBody" }
  | { kind: "invalidJson"; message: string }
  | { kind: "schemaMismatch"; issues: DecodeIssue[] }
  | { kind: "bodyReadFailed"; message: string }
  | { kind: "custom"; message: string; details?: unknown };

export type BodyPreview = {
  text: string;
  bytesRead: number;
  truncated: boolean;
};

export type TransportError =
  | { kind: "aborted" }
  | { kind: "timeout" }
  | { kind: "network"; cause?: unknown };

export type SendResult<R> =
  | { kind: "response"; response: R }
  | { kind: "transportError"; error: TransportError }
  | {
      kind: "decodeError";
      status: number;
      headers: Headers;
      error: DecodeError;
      preview: BodyPreview;
    }
  | {
      kind: "unhandledStatus";
      status: number;
      headers: Headers;
      preview: BodyPreview;
    };

export type TaggedEntry<T = unknown, Tag extends string = string> = {
  readonly tag: Tag;
  declare readonly _phantom: T;
  readonly _decode: unknown;
};

export type ResponseMap = Partial<Record<StatusMatcher, TaggedEntry>>;

export type InferResponseUnion<M extends ResponseMap> = {
  [K in keyof M]: M[K] extends TaggedEntry<infer T, infer Tag>
    ? { tag: Tag; body: T }
    : never;
}[keyof M];

export type RetryPolicy = {
  readonly methods?: readonly Method[];
  readonly maxAttempts?: number;
  readonly retryableStatuses?: readonly number[];
  readonly backoffMs?: { readonly initial: number; readonly max: number };
};

export type RequestSpec<Responses extends ResponseMap = ResponseMap> = {
  method: Method;
  path?: readonly (string | number)[];
  absoluteUrl?: string | URL;
  query?: Record<string, QueryValue | readonly QueryValue[] | undefined>;
  headers?: Record<string, string | undefined>;
  body?: unknown;
  responses: Responses;
  retry?: RetryPolicy;
  deadlineMs?: number;
};

export type ClientSpec = {
  baseUrl?: string | URL;
  headers?: Record<string, string | undefined>;
  query?: Record<string, QueryValue | readonly QueryValue[] | undefined>;
  responses?: ResponseMap;
  retry?: RetryPolicy;
  deadlineMs?: number;
  diagnostics?: {
    bodyPreviewBytes?: number;
  };
};
```

Key constraints honored:
- **D-01**: `TaggedEntry<T, Tag extends string>` carries `declare readonly _phantom: T` (type-only, no runtime slot) and `readonly _decode: unknown` (runtime decoder slot for Phase 4).
- **D-02**: `ResponseMap = Partial<Record<StatusMatcher, TaggedEntry>>`. `RequestSpec<Responses extends ResponseMap = ResponseMap>` carries a generic so inference flows.
- **D-03/D-04**: `InferResponseUnion<M>` maps each `TaggedEntry<T, Tag>` entry to `{ tag: Tag; body: T }`. The `extends TaggedEntry<infer T, infer Tag> ? ... : never` conditional filters out `undefined` from Partial entries under `noUncheckedIndexedAccess`.
- **D-05**: `RetryPolicy` defined with `methods`, `maxAttempts`, `retryableStatuses`, `backoffMs` as specified.
- **D-06**: `ClientSpec.query` included (SPEC prose gap — it must exist for merge semantics to work).
- **D-11**: Zero import statements. `Headers`, `URL` are ambient globals.
- **D-12**: `declare readonly _phantom: T` — not a value field. Under `exactOptionalPropertyTypes`, this prevents a phantom property slot from being required in object literals.
- Optional fields use `field?: T` not `field?: T | undefined` (exactOptionalPropertyTypes discipline).
- `body?: unknown` in RequestSpec — Phase 4 narrows this to the real `Body` type.

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Library consumer → types.ts | Callers construct RequestSpec values using the exported types; type system enforces shape at compile time |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01-01 | Tampering | `exactOptionalPropertyTypes` + object spread at consumer call sites | accept | This phase introduces the types; it does not contain merge logic. Spread anti-pattern is addressed in Plan 03's `mergeHeaders`/`mergeQuery` implementation. The type definitions themselves do not spread. |
| T-02-01-02 | Information Disclosure | `_decode: unknown` slot on TaggedEntry | accept | `_decode` is an opaque runtime slot. It holds a decoder function cast by Phase 4. Type system hides the concrete type from consumers — no sensitive information exposure at this layer. |
| T-02-01-03 | Tampering | `declare readonly _phantom: T` phantom field bypass | accept | The `declare` approach is purely type-level. A consumer CAN assign a non-matching `_phantom` value at runtime (TypeScript can't prevent runtime mutation), but there is no runtime enforcement needed here — phantom type is solely for compile-time inference of `R` in `Request<R>`. Phase 2 does not rely on `_phantom` at runtime. |

> **Security scope note:** This plan creates zero runtime logic — only type definitions and stub test files. No authentication, session management, cryptography, input validation, or external I/O surfaces are introduced. ASVS L1 categories V2–V4 and V6 do not apply. V5 (input validation) applies in Plan 02/03 where `encodeURIComponent` and `URLSearchParams` are used for path/query encoding.
</threat_model>
