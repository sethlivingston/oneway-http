    function buildUrlSearchParams(
      entries: Record<string, string | readonly string[]>,
    ): URLSearchParams {
      const params = new URLSearchParams();
      for (const [key, value] of Object.entries(entries)) {
        if (Array.isArray(value)) {
          for (const v of value) {
            params.append(key, v as string);
          }
        } else {
          params.append(key, value as string);
        }
      }
      return params;
    }

      const internal = fromBody(body);
      switch (internal.kind) {
        case "none":
          return {};

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| caller → Body.json(value) | Caller-provided value crosses into JSON.stringify; arbitrary JS values including attacker-influenced data |
| serializeBody → fetchInit.body | Serialized bytes cross into the HTTP transport layer |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-01-01 | Tampering | `Body.json(value)` — `value.toJSON()` method | accept | `.toJSON()` executes in caller's trust domain; library is not responsible for caller-provided values executing code during stringify. Document that `Body.json()` calls `JSON.stringify` at send time. |
| T-04-01-02 | Denial of Service | `serializeBody` — `JSON.stringify` on large objects | accept | Library is a client; the caller controls the request body. No size limit enforced at this layer — consistent with all major HTTP clients (ky, axios, node-fetch). |
| T-04-01-03 | Information Disclosure | `RequestError.bodySerializationFailed` message field | mitigate | Catch block in send.ts uses `e instanceof Error ? e.message : String(e)` — limits to Error.message only. Do NOT include stack trace or full error object in the structured result. |
</threat_model>
